## =====> Wed Sep 30 00:52:29 JST 2020 <=====

ファイル名を連番に変更

```bash
#!/bin/bash

SELF=$(cd $(dirname $0); pwd)

function _renban() {
    counter=1
    local list=$(ls $@/*.JPG 2>/dev/null)
    echo "list: ${list}"
    [ -z ${list} ] && return

    for file in ${list}; do
        local oldname=${SELF}/${file}
        local ext=${file##*.}
        local newname=${SELF}/$@/$(printf "%05d\n" ${counter}).${ext}

        mv ${oldname} ${newname}
        [ $? -eq 0 ] && echo "rename: ${oldname} to ${newname}"

        counter=$(expr ${counter} + 1)
    done
}

_renban $@

exit 0
```
